cmake_minimum_required(VERSION 3.20)
project(AstraScan VERSION 0.1.0 LANGUAGES CXX)
option(ASTRA_ENABLE_TLS "Build TLS inspection using OpenSSL" ON)
option(ASTRA_ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
find_package(Threads REQUIRED)
add_library(astra_core src/config.cpp src/scanner.cpp src/report.cpp)
target_compile_features(astra_core PUBLIC cxx_std_20)
target_include_directories(astra_core PUBLIC include)
target_link_libraries(astra_core PUBLIC Threads::Threads)
target_compile_options(astra_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
if(ASTRA_ENABLE_TLS)
  find_package(OpenSSL 3.0 REQUIRED)
  target_compile_definitions(astra_core PUBLIC ASTRA_TLS)
  target_link_libraries(astra_core PUBLIC OpenSSL::SSL OpenSSL::Crypto)
endif()
if(ASTRA_ENABLE_SANITIZERS)
  target_compile_options(astra_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(astra_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(astrascan src/main.cpp)
target_link_libraries(astrascan PRIVATE astra_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(unit_tests tests/unit.cpp)
  target_link_libraries(unit_tests PRIVATE astra_core)
  add_test(NAME unit COMMAND unit_tests)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:astrascan>)
  add_test(NAME diff COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_SOURCE_DIR}/tests -p test_*.py)
endif()
install(TARGETS astrascan RUNTIME DESTINATION bin)
install(PROGRAMS tools/astra-diff.py DESTINATION bin RENAME astra-diff)

install(PROGRAMS tools/astra-policy.py DESTINATION bin RENAME astra-policy)
